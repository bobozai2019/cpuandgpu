cmake_minimum_required(VERSION 3.24)
project(DX11ParallelClassroom VERSION 1.0 LANGUAGES CXX)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT 01_pipeline)
if(NOT WIN32)
  message(FATAL_ERROR "This teaching project requires Windows and the Windows SDK (Direct3D 11).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(classroom STATIC
  src/common/app.cpp src/common/gpu.cpp src/common/particles.cpp)
set(CLASSROOM_HEADERS
  "${CMAKE_CURRENT_SOURCE_DIR}/src/common/classroom.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/common/rolling_metrics.h")
target_sources(classroom PUBLIC ${CLASSROOM_HEADERS})
source_group("Head" FILES ${CLASSROOM_HEADERS})
target_include_directories(classroom PUBLIC src/common)
target_compile_definitions(classroom PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(classroom PUBLIC /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(classroom PUBLIC d3d11 dxgi d3dcompiler windowscodecs ole32 user32 gdi32)
set(CHAPTERS
  01_pipeline 02_timing 03_upload 04_cpu_jobs
  05_compute 06_readback 07_deferred 08_integrated)
foreach(CHAPTER IN LISTS CHAPTERS)
  add_executable(${CHAPTER} src/chapters/${CHAPTER}.cpp)
  target_link_libraries(${CHAPTER} PRIVATE classroom)
  set_target_properties(${CHAPTER} PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endforeach()
file(COPY shaders DESTINATION "${CMAKE_BINARY_DIR}")
file(GLOB COURSE_DOCS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/docs/*.md")
add_custom_target(course_materials SOURCES
  README.md docs/handbook.html shaders/particles.hlsl ${COURSE_DOCS})
enable_testing()
add_executable(rolling_metrics_test tests/rolling_metrics.cpp)
target_link_libraries(rolling_metrics_test PRIVATE classroom)
add_test(NAME rolling_metrics COMMAND rolling_metrics_test)
add_executable(comparison_test tests/comparison.cpp)
target_link_libraries(comparison_test PRIVATE classroom)
add_test(NAME comparison COMMAND comparison_test)
set_tests_properties(comparison PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(gui_controls_test tests/gui_controls.cpp)
target_link_libraries(gui_controls_test PRIVATE classroom)
foreach(CHAPTER_INDEX RANGE 1 8)
  add_test(NAME gui_controls_${CHAPTER_INDEX} COMMAND gui_controls_test ${CHAPTER_INDEX})
  set_tests_properties(gui_controls_${CHAPTER_INDEX} PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE
    WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
foreach(CHAPTER IN LISTS CHAPTERS)
  add_test(NAME ${CHAPTER}_warp COMMAND ${CHAPTER}
    --headless --warp --frames 48 --count 4096 --draws 256 --validate --debug)
  set_tests_properties(${CHAPTER}_warp PROPERTIES TIMEOUT 60 WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
