cmake_minimum_required(VERSION 3.24)
project(CpuParallelLab LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This lab uses Windows thread CPU accounting and a native Win32 viewer.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cpu_lab STATIC src/engine.cpp src/raytrace.cpp src/export.cpp)
target_include_directories(cpu_lab PUBLIC src)
target_compile_definitions(cpu_lab PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(cpu_lab PUBLIC /W4 /utf-8 /permissive- /EHsc /fp:precise)
add_executable(cpu_parallel_lab WIN32 src/gui.cpp)
target_link_libraries(cpu_parallel_lab PRIVATE cpu_lab user32 gdi32 shell32)
add_executable(cpu_benchmark src/benchmark.cpp)
target_link_libraries(cpu_benchmark PRIVATE cpu_lab)
add_executable(cpu_lab_tests tests/tests.cpp)
target_link_libraries(cpu_lab_tests PRIVATE cpu_lab)
enable_testing()
add_test(NAME concurrency_contracts COMMAND cpu_lab_tests)
set_tests_properties(concurrency_contracts PROPERTIES TIMEOUT 60)
add_test(NAME gui_contracts COMMAND cpu_parallel_lab --self-test)
set_tests_properties(gui_contracts PROPERTIES TIMEOUT 60)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT cpu_parallel_lab)
