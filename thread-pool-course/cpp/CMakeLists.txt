cmake_minimum_required(VERSION 3.24)
project(ThreadPoolCourse LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(thread_pool INTERFACE)
set(THREAD_POOL_HEADERS
  "${CMAKE_CURRENT_SOURCE_DIR}/include/thread_pool.hpp")
target_sources(thread_pool INTERFACE ${THREAD_POOL_HEADERS})
source_group("Head" FILES ${THREAD_POOL_HEADERS})
target_include_directories(thread_pool INTERFACE include)
target_compile_features(thread_pool INTERFACE cxx_std_17)
target_link_libraries(thread_pool INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(thread_pool INTERFACE /W4 /utf-8 /permissive- /EHsc)
else()
  target_compile_options(thread_pool INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(thread_pool_demo src/main.cpp)
target_link_libraries(thread_pool_demo PRIVATE thread_pool)
add_executable(thread_pool_tests tests/thread_pool_tests.cpp)
target_link_libraries(thread_pool_tests PRIVATE thread_pool)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT thread_pool_demo)
enable_testing()
add_test(NAME thread_pool_contracts COMMAND thread_pool_tests)
set_tests_properties(thread_pool_contracts PROPERTIES TIMEOUT 60)
